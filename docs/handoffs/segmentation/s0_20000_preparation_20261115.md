```bash
set -euo pipefail
cd /root/autodl-tmp/fashion-grounding/workspaces/segmentation
export S0_ROOT=/root/autodl-tmp/fashion-grounding
export S0_PY="$S0_ROOT/envs/segmentation-v1/bin/python"
export PYTHONPATH=src
: "${S0_RUN_ID:?Set a new unique run ID only when execution is authorized}"
export S0_OUT="$S0_ROOT/runs/segmentation/$S0_RUN_ID"
mkdir "$S0_OUT"
"$S0_PY" scripts/with_gpu_lock.py \
  --coordination "$S0_ROOT/coordination" --run-id "$S0_RUN_ID" \
  --output "$S0_OUT" --seconds 10795 -- \
  /usr/bin/timeout --signal=KILL 10790s bash -euo pipefail \
  >"$S0_OUT/stage.log" 2>&1 <<'S0_STAGE'
"$S0_PY" -m fashion_grounding.segmentation.train \
  --package "$S0_ROOT/shared/processed/fp_train5000_v1" \
  --source "$S0_ROOT/vendor/segmentation/Mask2Former" \
  --resume "$S0_ROOT/runs/segmentation/segmentation-resume5000-train-20261113-r01/checkpoint_005000.pth" \
  --resume-extension configs/runs/segmentation/resume_5000_to_20000.json \
  --config configs/runs/segmentation/s0_resume_20000.json --output "$S0_OUT/train"
points=(6000 8000 10000 12000 14000 16000 18000 20000)
checkpoints=()
reports=()
for u in "${points[@]}"; do
  printf -v padded '%06d' "$u"
  checkpoints+=("$S0_OUT/train/checkpoint_$padded.pth")
  reports+=("$S0_OUT/dev/update_$padded/evaluation_report.json")
done
"$S0_PY" -m fashion_grounding.segmentation.evaluate \
  --package "$S0_ROOT/shared/processed/fp_dev250_v1" \
  --training-package "$S0_ROOT/shared/processed/fp_train5000_v1" \
  --source "$S0_ROOT/vendor/segmentation/Mask2Former" \
  --config configs/runs/segmentation/dev_ap_250_resume_20000.json \
  --output "$S0_OUT/dev" --checkpoints "${checkpoints[@]}"
"$S0_PY" -m fashion_grounding.segmentation.select_checkpoint \
  --reports "${reports[@]}" --output "$S0_OUT/new_budget_selection"
S0_STAGE
```
